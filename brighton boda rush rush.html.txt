<script>
let player=document.getElementById('player');
let obs=document.getElementById('obstacle');
let coin=document.getElementById('coin');
let scoreBox=document.getElementById('score');
let coins=0, score=0, bottom=85, jumping=false, gameOver=false;

function jump(){
 if(jumping||gameOver) return;
 jumping=true;
 let up=setInterval(()=>{
   bottom+=9; player.style.bottom=bottom+'px';
   if(bottom>=200){clearInterval(up);
     let down=setInterval(()=>{
       bottom-=8; player.style.bottom=bottom+'px';
       if(bottom<=85){clearInterval(down);bottom=85;jumping=false;}
     },20);
   }
 },20);
}
function resetObs(){obs.style.left='450px';}
function resetCoin(){coin.style.left='500px'; coin.style.bottom=(130+Math.random()*70)+'px';}
function updateScore(){scoreBox.innerHTML='Coins: '+coins+' | Score: '+score;}

let gameLoop=setInterval(()=>{
 if(gameOver) return;
 obs.style.left=(obs.offsetLeft-6)+'px';
 if(obs.offsetLeft<-50){resetObs(); score++; updateScore();}
 coin.style.left=(coin.offsetLeft-5)+'px';
 if(coin.offsetLeft<-40){resetCoin();}
 let oL=obs.offsetLeft, cL=coin.offsetLeft, pB=bottom;
 if(oL>30 && oL<100 && pB<125){gameOver=true; scoreBox.innerHTML='GAME OVER! Coins:'+coins+' | Refresh to restart';}
 if(cL>30 && cL<105 && pB> (parseInt(coin.style.bottom)-90) && pB < (parseInt(coin.style.bottom)-40)){
   coins++; score+=5; resetCoin(); updateScore();
   player.style.transform='scale(1.3)'; setTimeout(()=>player.style.transform='scale(1)',150);
 }
},22);
resetObs(); resetCoin();
</script>
